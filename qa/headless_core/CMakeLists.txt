# Standalone build of KiCad's toolkit-independent libraries (core_base, kimath, sexpr) with
# NO wxWidgets: wx is never searched for, and any <wx/...> include fails to compile.
#
# It reuses the libraries' own CMakeLists, so it doubles as the regression check for step 1 of
# the wx removal, and as the recipe for non-wx front ends (e.g. a Qt UI) that want KiCad's
# geometry and s-expression code.
#
#   cmake -S qa/headless_core -B build-headless -G Ninja && cmake --build build-headless
#   build-headless/headless_core_demo path/to/file.kicad_sch

cmake_minimum_required( VERSION 3.21 )
project( kicad_headless_core C CXX )

set( CMAKE_CXX_STANDARD 20 )
set( CMAKE_CXX_STANDARD_REQUIRED ON )

get_filename_component( KICAD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE )

if( DEFINED wxWidgets_FOUND OR DEFINED wxWidgets_LIBRARIES )
    message( FATAL_ERROR "wxWidgets must not be part of the headless core build" )
endif()

# Minimal config.h (the full one is generated by KiCad's top-level CMakeLists)
include( CheckSymbolExists )
check_symbol_exists( clock_gettime "time.h" HAVE_CLOCK_GETTIME )
check_symbol_exists( gettimeofday "sys/time.h" HAVE_GETTIMEOFDAY_FUNC )
configure_file( config.h.in ${CMAKE_BINARY_DIR}/config.h )

find_package( Boost REQUIRED )

set( FASTFLOAT_INSTALL OFF CACHE BOOL "" FORCE )
set( FMT_INSTALL OFF CACHE BOOL "" FORCE )

# Third party code used by the libraries
add_subdirectory( ${KICAD_ROOT}/thirdparty/clipper2   thirdparty/clipper2 )
add_subdirectory( ${KICAD_ROOT}/thirdparty/rtree      thirdparty/rtree )
add_subdirectory( ${KICAD_ROOT}/thirdparty/fmt        thirdparty/fmt )
add_subdirectory( ${KICAD_ROOT}/thirdparty/fast_float thirdparty/fast_float )

# The libraries, through their own CMakeLists (single binary: process state can be static)
set( KICORE_RT_SHARED OFF CACHE BOOL "" FORCE )
add_subdirectory( ${KICAD_ROOT}/libs/core   libs/core )
add_subdirectory( ${KICAD_ROOT}/libs/kimath libs/kimath )
add_subdirectory( ${KICAD_ROOT}/libs/sexpr  libs/sexpr )

# Header-only helpers still shared from the top-level include/ (base_units.h, core/mirror.h)
target_include_directories( kimath PUBLIC ${KICAD_ROOT}/include ${CMAKE_BINARY_DIR} )
target_include_directories( sexpr PRIVATE ${KICAD_ROOT}/include )

add_executable( headless_core_demo demo.cpp )
target_link_libraries( headless_core_demo core_base kimath sexpr )
target_compile_definitions( headless_core_demo PRIVATE KICAD_NO_WX )
